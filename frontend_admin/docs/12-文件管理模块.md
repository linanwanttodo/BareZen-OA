## 12.2 文件管理页面 src/views/file/index.vue

```vue
<template>
  <div class="page-container">
    <div class="search-bar">
      <el-form inline :model="queryParams">
        <el-form-item label="文件名称">
          <el-input v-model="queryParams.fileName" placeholder="请输入文件名称" clearable />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="handleSearch">搜索</el-button>
          <el-button @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </div>

    <div class="toolbar">
      <el-upload
        :show-file-list="false"
        :http-request="handleUpload"
        accept="*"
      >
        <el-button type="primary"><el-icon><Upload /></el-icon> 上传文件</el-button>
      </el-upload>
    </div>

    <el-table v-loading="loading" :data="list">
      <el-table-column label="文件名" prop="fileName" min-width="200" show-overflow-tooltip />
      <el-table-column label="文件大小" prop="fileSize" width="120" align="center">
        <template #default="{ row }">
          {{ formatFileSize(row.fileSize) }}
        </template>
      </el-table-column>
      <el-table-column label="文件类型" prop="fileType" width="120" />
      <el-table-column label="上传时间" prop="createTime" width="170" />
      <el-table-column label="操作" width="180" fixed="right" align="center">
        <template #default="{ row }">
          <el-button type="primary" link @click="handleDownload(row)">下载</el-button>
          <el-button type="danger" link @click="handleDelete(row)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <Pagination v-model:page="queryParams.pageNum" v-model:limit="queryParams.pageSize" :total="total" @pagination="loadData" />
  </div>
</template>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Upload } from '@element-plus/icons-vue'
import Pagination from '@/components/Pagination.vue'
import { usePage } from '@/composables/usePage'
import { useDownload } from '@/composables/useDownload'
import { getFileList, deleteFile, uploadFile } from '@/api/file'
import type { FileInfo } from '@/api/file'

const { loading, list, total, loadData } = usePage<FileInfo>()
const { downloadById } = useDownload()
const queryParams = reactive({ pageNum: 1, pageSize: 10, fileName: '' })

function formatFileSize(size: number): string {
  if (size < 1024) return `${size} B`
  if (size < 1024 * 1024) return `${(size / 1024).toFixed(1)} KB`
  if (size < 1024 * 1024 * 1024) return `${(size / 1024 / 1024).toFixed(1)} MB`
  return `${(size / 1024 / 1024 / 1024).toFixed(1)} GB`
}

function handleSearch() { queryParams.pageNum = 1; loadData(getFileList) }
function handleReset() { Object.assign(queryParams, { pageNum: 1, pageSize: 10, fileName: '' }); loadData(getFileList) }

async function handleUpload(options: any) {
  const file = options.file
  try {
    await uploadFile(file)
    ElMessage.success('上传成功')
    loadData(getFileList)
  } catch {
    ElMessage.error('上传失败')
  }
}

function handleDownload(row: FileInfo) {
  downloadById(row.id, row.fileName)
}

async function handleDelete(row: FileInfo) {
  await ElMessageBox.confirm(`确定删除文件 "${row.fileName}" 吗？`, '警告', { type: 'warning' })
  await deleteFile(row.id)
  ElMessage.success('删除成功')
  loadData(getFileList)
}

onMounted(() => { loadData(getFileList) })
</script>
```
