## 15.2 在线用户页面 src/views/monitor/online/index.vue

```vue
<template>
  <div class="page-container">
    <div class="toolbar flex-between">
      <div class="text-sm text-gray-500">
        共 <span class="font-bold text-primary-600">{{ total }}</span> 人在线
      </div>
      <el-button @click="loadData" :loading="loading">
        <el-icon><Refresh /></el-icon> 刷新
      </el-button>
    </div>

    <el-table v-loading="loading" :data="list" stripe>
      <el-table-column label="用户名" prop="userName" width="150" />
      <el-table-column label="昵称" prop="nickName" width="150" />
      <el-table-column label="登录IP" prop="ip" width="150" />
      <el-table-column label="登录时间" prop="loginTime" min-width="180" />
      <el-table-column label="最后访问" prop="lastAccessTime" min-width="180" />
      <el-table-column label="操作" width="120" fixed="right" align="center">
        <template #default="{ row }">
          <el-button type="danger" link @click="handleForceOffline(row)">
            强制下线
          </el-button>
        </template>
      </el-table-column>
    </el-table>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Refresh } from '@element-plus/icons-vue'
import { getOnlineList, forceOffline } from '@/api/online'
import type { OnlineUser } from '@/api/online'

const loading = ref(false)
const list = ref<OnlineUser[]>([])
const total = ref(0)
let refreshTimer: number | undefined

async function loadData() {
  loading.value = true
  try {
    const { data } = await getOnlineList()
    list.value = data || []
    total.value = list.value.length
  } finally {
    loading.value = false
  }
}

async function handleForceOffline(row: OnlineUser) {
  await ElMessageBox.confirm(`确定强制下线用户 "${row.nickName}" 吗？`, '警告', { type: 'warning' })
  await forceOffline(row.token)
  ElMessage.success('已强制下线')
  loadData()
}

onMounted(() => {
  loadData()
  // 每 30 秒自动刷新
  refreshTimer = window.setInterval(loadData, 30 * 1000)
})

onUnmounted(() => {
  if (refreshTimer) clearInterval(refreshTimer)
})
</script>
```
