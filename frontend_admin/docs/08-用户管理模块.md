## 8.2 用户管理主页面 src/views/system/user/index.vue

```vue
<template>
  <div class="page-container">
    <!-- 搜索栏 -->
    <div class="search-bar">
      <el-form inline :model="queryParams" label-width="80px">
        <el-form-item label="用户名称">
          <el-input
            v-model="queryParams.userName"
            placeholder="请输入用户名称"
            clearable
            @keyup.enter="handleSearch"
          />
        </el-form-item>

        <el-form-item label="手机号码">
          <el-input
            v-model="queryParams.phonenumber"
            placeholder="请输入手机号"
            clearable
            @keyup.enter="handleSearch"
          />
        </el-form-item>

        <el-form-item label="状态">
          <el-select v-model="queryParams.status" placeholder="全部" clearable style="width: 150px">
            <el-option label="正常" :value="0" />
            <el-option label="停用" :value="1" />
          </el-select>
        </el-form-item>

        <el-form-item label="创建时间">
          <el-date-picker
            v-model="dateRange"
            type="daterange"
            range-separator="至"
            start-placeholder="开始日期"
            end-placeholder="结束日期"
            value-format="YYYY-MM-DD"
            style="width: 240px"
          />
        </el-form-item>

        <el-form-item>
          <el-button type="primary" @click="handleSearch">
            <el-icon><Search /></el-icon> 搜索
          </el-button>
          <el-button @click="handleReset">
            <el-icon><Refresh /></el-icon> 重置
          </el-button>
        </el-form-item>
      </el-form>
    </div>

    <!-- 操作栏 -->
    <div class="toolbar">
      <el-button type="primary" v-permission="['system:user:add']" @click="handleAdd">
        <el-icon><Plus /></el-icon> 新增
      </el-button>
      <el-button
        type="danger"
        v-permission="['system:user:del']"
        :disabled="selectedRows.length === 0"
        @click="handleBatchDelete"
      >
        <el-icon><Delete /></el-icon> 批量删除
      </el-button>
    </div>

    <!-- 表格 -->
    <el-table
      v-loading="loading"
      :data="list"
      stripe
      @selection-change="handleSelectionChange"
    >
      <el-table-column type="selection" width="50" align="center" />

      <el-table-column label="用户名称" prop="userName" width="120" />
      <el-table-column label="用户昵称" prop="nickName" width="120" />
      <el-table-column label="性别" prop="sex" width="80">
        <template #default="{ row }">
          <el-tag :type="row.sex === 1 ? '' : row.sex === 2 ? 'success' : 'info'">
            {{ sexMap[row.sex] }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column label="手机号码" prop="phone" width="120" />
      <el-table-column label="邮箱" prop="email" min-width="180" show-overflow-tooltip />
      <el-table-column label="状态" prop="status" width="80" align="center">
        <template #default="{ row }">
          <el-switch
            v-model="row.status"
            :active-value="0"
            :inactive-value="1"
            v-permission="['system:user:edit']"
            @change="handleStatusChange(row)"
          />
        </template>
      </el-table-column>
      <el-table-column label="创建时间" prop="createTime" width="170" />
      <el-table-column label="操作" width="200" fixed="right" align="center">
        <template #default="{ row }">
          <el-button type="primary" link v-permission="['system:user:edit']" @click="handleEdit(row)">
            编辑
          </el-button>
          <el-button type="danger" link v-permission="['system:user:del']" @click="handleDelete(row)">
            删除
          </el-button>
          <el-dropdown v-permission="['system:user:resetPwd']" @command="(cmd: string) => handleCommand(cmd, row)">
            <el-button type="info" link>
              更多 <el-icon><ArrowDown /></el-icon>
            </el-button>
            <template #dropdown>
              <el-dropdown-menu>
                <el-dropdown-item command="resetPwd">重置密码</el-dropdown-item>
              </el-dropdown-menu>
            </template>
          </el-dropdown>
        </template>
      </el-table-column>
    </el-table>

    <!-- 分页 -->
    <Pagination
      v-model:page="queryParams.pageNum"
      v-model:limit="queryParams.pageSize"
      :total="total"
      @pagination="loadData"
    />

    <!-- 新增/编辑弹窗 -->
    <UserForm v-model="formVisible" :data="currentUser" @success="loadData" />
  </div>
</template>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Search, Refresh, Plus, Delete, ArrowDown } from '@element-plus/icons-vue'
import Pagination from '@/components/Pagination.vue'
import { usePage } from '@/composables/usePage'
import { getUserList, deleteUser, batchDeleteUsers, changeUserStatus } from '@/api/user'
import type { UserInfo, UserQuery } from '@/types/user'
import UserForm from './components/UserForm.vue'

const { loading, list, total, loadData } = usePage<UserInfo>()

const queryParams = reactive<UserQuery>({
  pageNum: 1,
  pageSize: 10,
  userName: '',
  phonenumber: '',
  status: undefined
})

const dateRange = ref<[string, string]>()
const selectedRows = ref<UserInfo[]>([])
const formVisible = ref(false)
const currentUser = ref<UserInfo | null>(null)

const sexMap: Record<number, string> = {
  0: '未知',
  1: '男',
  2: '女'
}

function handleSearch() {
  queryParams.pageNum = 1
  loadData(getUserList)
}

function handleReset() {
  queryParams.pageNum = 1
  queryParams.userName = ''
  queryParams.phonenumber = ''
  queryParams.status = undefined
  dateRange.value = undefined
  loadData(getUserList)
}

function handleSelectionChange(selection: UserInfo[]) {
  selectedRows.value = selection
}

function handleAdd() {
  currentUser.value = null
  formVisible.value = true
}

function handleEdit(row: UserInfo) {
  currentUser.value = { ...row }
  formVisible.value = true
}

async function handleDelete(row: UserInfo) {
  await ElMessageBox.confirm(`确定删除用户 "${row.nickName}" 吗？`, '警告', { type: 'warning' })
  await deleteUser(row.userId)
  ElMessage.success('删除成功')
  loadData(getUserList)
}

async function handleBatchDelete() {
  const count = selectedRows.value.length
  await ElMessageBox.confirm(`确定删除选中的 ${count} 个用户吗？`, '警告', { type: 'warning' })
  await batchDeleteUsers({ ids: selectedRows.value.map((r) => r.userId) })
  ElMessage.success(`成功删除 ${count} 个用户`)
  selectedRows.value = []
  loadData(getUserList)
}

async function handleStatusChange(row: UserInfo) {
  const action = row.status === 0 ? '启用' : '停用'
  try {
    await changeUserStatus(row.userId, row.status)
    ElMessage.success(`${action}成功`)
  } catch {
    row.status = row.status === 0 ? 1 : 0
  }
}

function handleCommand(cmd: string, row: UserInfo) {
  if (cmd === 'resetPwd') {
    ElMessage.info(`重置密码功能（用户 ID: ${row.userId}）`)
  }
}

onMounted(() => {
  loadData(getUserList)
})
</script>

<style lang="scss" scoped>
.search-bar {
  background: #fff;
  padding: 18px 16px 0;
  border-radius: 4px;
  margin-bottom: 12px;
}

.toolbar {
  background: #fff;
  padding: 12px 16px;
  border-radius: 4px;
  margin-bottom: 12px;
}
</style>
```
