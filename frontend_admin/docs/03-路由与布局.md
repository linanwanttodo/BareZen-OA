## 3.1 Vue Router 配置

### 3.1.1 路由实例 src/router/index.ts

路由的核心配置，包含静态路由和动态路由的注册：

```typescript
import { createRouter, createWebHistory, type RouteRecordRaw } from 'vue-router'
import type { AppRouteRecordRaw } from '@/types/router'

// 静态路由（无需权限）
const staticRoutes: AppRouteRecordRaw[] = [
  {
    path: '/login',
    component: () => import('@/views/login/index.vue'),
    meta: { title: '登录', hidden: true }
  },
  {
    path: '/404',
    component: () => import('@/views/error/404.vue'),
    meta: { title: '404', hidden: true }
  },
  {
    path: '/:pathMatch(.*)*',
    redirect: '/404',
    meta: { hidden: true }
  }
]

// 动态路由（需要权限，从后端获取）
const dynamicRoutes: AppRouteRecordRaw[] = []

// 创建路由实例
const router = createRouter({
  // 使用 HTML5 History 模式
  history: createWebHistory(import.meta.env.VITE_APP_BASE_API),
  routes: staticRoutes,
  // 路由滚动行为
  scrollBehavior: () => ({ left: 0, top: 0 })
})

// 添加动态路由的辅助函数
export function addRoutes(routes: AppRouteRecordRaw[]) {
  routes.forEach((route) => {
    // 避免重复注册
    if (!router.getRoutes().find((r) => r.path === route.path)) {
      router.addRoute(route)
    }
  })
}

// 重置路由（登出时使用）
export function resetRouter() {
  const routes = router.getRoutes()
  routes.forEach((route) => {
    if (!staticRoutes.find((r) => r.path === route.path)) {
      router.removeRoute(route.name as string)
    }
  })
}

export default router
```

### 3.1.3 动态路由 src/router/routes/async.ts

动态路由生成（从后端菜单 API 获取并转换为 Vue Router 格式）：

```typescript
import type { AppRouteRecordRaw } from '@/types/router'
import type { MenuItem } from '@/types/menu'
import { addRoutes } from '@/router'

/**
 * 将后端菜单数据转换为 Vue Router 格式
 *
 * @param menus - 后端返回的菜单列表（树形结构）
 * @param parentPath - 父级路径（用于拼接路径）
 * @returns Vue Router 格式的路由数组
 */
export function generateRoutes(menus: MenuItem[]): AppRouteRecordRaw[] {
  const routes: AppRouteRecordRaw[] = []

  menus.forEach((menu) => {
    const route: AppRouteRecordRaw = {
      path: menu.path,
      name: menu.path.replace(/\//g, '-'),      // /system/user -> system-user
      component: loadComponent(menu.component),
      meta: {
        title: menu.menuName,
        icon: menu.icon,
        hidden: menu.visible === 1,
        noCache: menu.isCache === 1,
        link: menu.isFrame === 0 ? menu.path : undefined
      }
    }

    // 递归处理子菜单
    if (menu.children && menu.children.length > 0) {
      // 如果是一级菜单且有子菜单，需要设置 alwaysShow
      route.meta = route.meta || {}
      route.meta.alwaysShow = true
      route.children = generateRoutes(menu.children, menu.path)
    }

    routes.push(route)
  })

  return routes
}

/**
 * 加载组件
 * - 动态路由使用 () => import() 实现路由懒加载
 * - layouts 下的组件不需要懒加载（全局复用）
 *
 * @param componentPath - 组件路径，如 @/views/system/user/index.vue
 */
function loadComponent(componentPath?: string) {
  if (!componentPath) {
    // 无组件路径的路由使用 Layout 作为占位
    return () => import('@/layout/index.vue')
  }

  // 如果是布局组件，直接加载
  if (componentPath.includes('Layout')) {
    return () => import('@/layout/index.vue')
  }

  // 业务组件：懒加载
  return () => import(/* @vite-ignore */ `@/${componentPath}`)
}

/**
 * 获取并注册动态路由
 * 由路由守卫在登录成功后调用
 */
export async function buildDynamicRoutes(): Promise<AppRouteRecordRaw[]> {
  // 从 API 获取用户的菜单权限
  const { data } = await import('@/api/menu').then((m) => m.getRouters())
  const routes = generateRoutes(data || [])

  // 添加到路由表
  addRoutes(routes)

  return routes
}
```

## 3.2 路由守卫

### 3.2.1 src/router/guard.ts

路由守卫的核心逻辑（登录校验 + 权限控制）：

```typescript
import router from './index'
import { getToken } from '@/utils/auth'
import { ElMessage } from 'element-plus'
import type { LocationQueryRaw } from 'vue-router'
import { buildDynamicRoutes } from './routes/async'

// 白名单：无需登录即可访问的路由
const whiteList = ['/login', '/404', '/401']

router.beforeEach(async (to, from, next) => {
  // 设置页面标题
  document.title = to.meta.title ? `${to.meta.title} - BareZen` : 'BareZen 管理后台'

  // 从 Cookie 获取 Token
  const hasToken = getToken()

  if (hasToken) {
    // 已登录
    if (to.path === '/login') {
      // 已登录访问登录页 -> 跳转到首页
      next({ path: '/' })
    } else {
      // 检查是否已有用户信息
      const hasUserInfo = !!localStorage.getItem('user_info')
      if (hasUserInfo) {
        next()
      } else {
        try {
          // 获取用户信息和动态路由
          await buildDynamicRoutes()
          // 修复：确保路由已注册后再导航
          if (to.matched.length === 0) {
            // 如果当前路由无法匹配（动态路由未注册），重新跳转
            next({ ...to, replace: true })
          } else {
            next()
          }
        } catch (error) {
          console.error('路由加载失败:', error)
          ElMessage.error('获取路由信息失败，请重新登录')
          next('/login')
        }
      }
    }
  } else {
    // 未登录
    if (whiteList.includes(to.path)) {
      // 白名单直接通过
      next()
    } else {
      // 其他页面重定向到登录页，携带原路径
      next(`/login?redirect=${to.path}`)
    }
  }
})

// 后置守卫：处理 404
router.afterEach((to) => {
  if (to.matched.length === 0) {
    // 如果所有父路由都无法匹配，说明是 404
    next({ path: '/404' })
  }
})
```

> **原理**：路由守卫在每次路由切换前执行。白名单内的路径（如登录页）可以直接访问；其他路径如果无 Token 则重定向到登录页。如果有 Token 但还没有加载用户信息，则从后端获取用户菜单权限并注册动态路由。

### 3.3.3 菜单项组件 src/layout/components/SidebarItem.vue

递归渲染菜单项（支持多级菜单）：

```vue
<template>
  <!-- 叶子菜单（无子菜单） -->
  <el-menu-item v-if="!item.children || item.children.length === 0" :index="resolvePath(item.path)">
    <el-icon><component :is="item.meta?.icon || 'Folder'" /></el-icon>
    <template #title>
      <span>{{ item.meta?.title }}</span>
    </template>
  </el-menu-item>

  <!-- 有子菜单：递归渲染 -->
  <el-sub-menu v-else :index="resolvePath(item.path)">
    <template #title>
      <el-icon><component :is="item.meta?.icon || 'Folder'" /></el-icon>
      <span>{{ item.meta?.title }}</span>
    </template>
    <SidebarItem
      v-for="child in item.children"
      :key="child.path"
      :item="child"
      :base-path="resolvePath(child.path)"
    />
  </el-sub-menu>
</template>

<script setup lang="ts">
import { isExternal } from '@/utils/validate'
import type { AppRouteRecordRaw } from '@/types/router'

interface Props {
  item: AppRouteRecordRaw
  basePath: string
}

const props = defineProps<Props>()

/**
 * 解析路由路径
 * - 外链直接返回完整 URL
 * - 内部路由拼接父级路径
 */
function resolvePath(path: string): string {
  if (isExternal(path)) {
    return path
  }
  // 只处理路径拼接，避免重复父级路径
  if (path.startsWith('/')) {
    return path
  }
  return `${props.basePath}/${path}`.replace(/\/+/g, '/')
}
</script>
```
