## 13.2 通知公告页面 src/views/notice/index.vue

```vue
<template>
  <div class="page-container">
    <div class="search-bar">
      <el-form inline :model="queryParams">
        <el-form-item label="公告标题">
          <el-input v-model="queryParams.noticeTitle" placeholder="请输入公告标题" clearable />
        </el-form-item>
        <el-form-item label="公告类型">
          <el-select v-model="queryParams.noticeType" placeholder="全部" clearable style="width: 150px">
            <el-option label="通知" :value="1" />
            <el-option label="公告" :value="2" />
          </el-select>
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="handleSearch">搜索</el-button>
          <el-button @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </div>

    <div class="toolbar">
      <el-button type="primary" v-permission="['system:notice:add']" @click="handleAdd">
        <el-icon><Plus /></el-icon> 新增
      </el-button>
    </div>

    <el-table v-loading="loading" :data="list" stripe>
      <el-table-column label="公告标题" prop="noticeTitle" min-width="200" show-overflow-tooltip />
      <el-table-column label="公告类型" prop="noticeType" width="100" align="center">
        <template #default="{ row }">
          <el-tag :type="row.noticeType === 1 ? 'success' : 'warning'">
            {{ row.noticeType === 1 ? '通知' : '公告' }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column label="状态" prop="status" width="100" align="center">
        <template #default="{ row }">
          <el-tag :type="row.status === 0 ? 'success' : 'info'">
            {{ row.status === 0 ? '已发布' : '草稿' }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column label="创建时间" prop="createTime" width="170" />
      <el-table-column label="操作" width="280" fixed="right" align="center">
        <template #default="{ row }">
          <el-button type="primary" link v-permission="['system:notice:edit']" @click="handleEdit(row)">
            编辑
          </el-button>
          <el-button
            v-if="row.status === 1"
            type="success"
            link
            v-permission="['system:notice:edit']"
            @click="handlePublish(row)"
          >
            发布
          </el-button>
          <el-button
            v-else
            type="warning"
            link
            v-permission="['system:notice:edit']"
            @click="handleRevoke(row)"
          >
            下架
          </el-button>
          <el-button type="danger" link v-permission="['system:notice:del']" @click="handleDelete(row)">
            删除
          </el-button>
        </template>
      </el-table-column>
    </el-table>

    <Pagination v-model:page="queryParams.pageNum" v-model:limit="queryParams.pageSize" :total="total" @pagination="loadData" />

    <!-- 公告表单弹窗 -->
    <NoticeForm v-model="formVisible" :data="currentNotice" @success="loadData" />
  </div>
</template>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Plus } from '@element-plus/icons-vue'
import Pagination from '@/components/Pagination.vue'
import { usePage } from '@/composables/usePage'
import { getNoticeList, deleteNotice, publishNotice, revokeNotice } from '@/api/notice'
import type { NoticeInfo } from '@/api/notice'
import NoticeForm from './components/NoticeForm.vue'

const { loading, list, total, loadData } = usePage<NoticeInfo>()
const queryParams = reactive({ pageNum: 1, pageSize: 10, noticeTitle: '', noticeType: undefined })
const formVisible = ref(false)
const currentNotice = ref<NoticeInfo | null>(null)

function handleSearch() { queryParams.pageNum = 1; loadData(getNoticeList) }
function handleReset() { Object.assign(queryParams, { pageNum: 1, pageSize: 10, noticeTitle: '', noticeType: undefined }); loadData(getNoticeList) }
function handleAdd() { currentNotice.value = null; formVisible.value = true }
function handleEdit(row: NoticeInfo) { currentNotice.value = { ...row }; formVisible.value = true }

async function handleDelete(row: NoticeInfo) {
  await ElMessageBox.confirm(`确定删除公告 "${row.noticeTitle}" 吗？`)
  await deleteNotice(row.id)
  ElMessage.success('删除成功')
  loadData(getNoticeList)
}

async function handlePublish(row: NoticeInfo) {
  await publishNotice(row.id)
  ElMessage.success('发布成功')
  loadData(getNoticeList)
}

async function handleRevoke(row: NoticeInfo) {
  await revokeNotice(row.id)
  ElMessage.success('下架成功')
  loadData(getNoticeList)
}

onMounted(() => { loadData(getNoticeList) })
</script>
```
