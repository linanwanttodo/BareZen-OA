## 6.1 权限控制全链路

权限控制分为三层：

```
┌─────────────────────────────────────────────────────────┐
│  1. 路由层   │ 路由守卫检查 Token，无 Token → 登录页      │
│             │ 动态路由根据后端返回的菜单生成                │
├─────────────────────────────────────────────────────────┤
│  2. 视图层   │ v-permission 指令控制按钮/区域的显示隐藏    │
│             │ 菜单根据 roles/perms 过滤                  │
├─────────────────────────────────────────────────────────┤
│  3. 接口层   │ 后端 @RequirePermission AOP 拦截           │
│             │ 前端 Token 携带，响应拦截 403               │
└─────────────────────────────────────────────────────────┘
```

```vue
  } finally {
```

## 6.3 路由守卫完善 src/router/guard.ts

在第 03 章的基础上完善，添加用户信息获取和错误处理：

```typescript
import router from './index'
import { getToken } from '@/utils/auth'
import { ElMessage } from 'element-plus'
import { buildDynamicRoutes } from '@/router/routes/async'
import { useUserStore } from '@/stores/user'

const whiteList = ['/login', '/404', '/401']

router.beforeEach(async (to, from, next) => {
  document.title = to.meta.title ? `${to.meta.title} - BareZen` : 'BareZen 管理后台'

  const hasToken = getToken()

  if (hasToken) {
    if (to.path === '/login') {
      next({ path: '/' })
    } else {
      const userStore = useUserStore()
      if (!userStore.userId) {
        try {
          // 获取用户信息
          await userStore.getUserInfo()
          // 加载动态路由
          await buildDynamicRoutes()
          // 修复动态路由注册时机问题
          next({ ...to, replace: true })
        } catch (error) {
          console.error('初始化用户信息失败:', error)
          ElMessage.error('初始化失败，请重新登录')
          next('/login')
        }
      } else {
        next()
      }
    }
  } else {
    if (whiteList.includes(to.path)) {
      next()
    } else {
      next(`/login?redirect=${to.path}`)
    }
  }
})
```

## 6.4 v-permission 指令 src/directive/permission.ts

按钮级别的权限控制：

```typescript
import type { Directive, DirectiveBinding } from 'vue'
import { usePermissionStore } from '@/stores/permission'

/**
 * v-permission 指令
 * 用法：
 *   <el-button v-permission="['system:user:add']">新增</el-button>
 *   <el-button v-permission="['system:user:edit', 'system:user:del']">操作</el-button>
 */
export const permission: Directive = {
  mounted(el: HTMLElement, binding: DirectiveBinding<string[]>) {
    const { value } = binding
    const permissionStore = usePermissionStore()

    if (value && value instanceof Array && value.length > 0) {
      const hasPermission = value.some((perm) => permissionStore.hasPermission(perm))
      if (!hasPermission) {
        el.parentNode?.removeChild(el)
      }
    } else {
      throw new Error('v-permission 指令需要传入权限标识数组')
    }
  }
}
```

注册指令到全局（在 main.ts 中）：

```typescript
// main.ts 中添加
import { permission } from '@/directive/permission'

app.directive('permission', permission)
```

## 6.5 权限工具函数 src/utils/permission.ts

```typescript
import { usePermissionStore } from '@/stores/permission'

/**
 * 检查是否有指定权限（供 JS 代码调用）
 */
export function hasPermission(perm: string | string[]): boolean {
  return usePermissionStore().hasPermission(perm)
}

/**
 * 获取用户权限标识列表
 */
export function getPermissions(): string[] {
  return usePermissionStore().permissions
}
```

```typescript
// 完整登出流程（在 Navbar.vue 中使用）
async function handleLogout() {
  await ElMessageBox.confirm('确定退出登录吗？', '提示', { type: 'warning' })

  try {
    await userStore.logout()
    // 重置路由
    usePermissionStore().resetPermission()
    // 跳转登录页
    router.push('/login')
  } catch (error) {
    // 忽略
  }
}
```
