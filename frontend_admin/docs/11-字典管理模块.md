## 11.3 useDict Hook src/composables/useDict.ts

通用字典数据获取 Hook，避免在每个页面重复调用：

```typescript
import { ref, onMounted } from 'vue'
import { getDictData } from '@/api/dict'
import type { DictData } from '@/api/dict'

/**
 * 获取指定类型的字典数据
 */
export function useDict(types: string[]) {
  const dictData = ref<Record<string, DictData[]>>({})
  const loading = ref(false)

  async function loadDictData() {
    loading.value = true
    try {
      for (const type of types) {
        const { data } = await getDictData(type)
        dictData.value[type] = data || []
      }
    } finally {
      loading.value = false
    }
  }

  /**
   * 根据类型和值获取标签
   */
  function getDictLabel(type: string, value: string): string {
    const items = dictData.value[type] || []
    const item = items.find((i) => i.dictValue === value)
    return item?.dictLabel || value
  }

  onMounted(() => {
    if (types.length > 0) {
      loadDictData()
    }
  })

  return {
    dictData,
    loading,
    loadDictData,
    getDictLabel
  }
}
```
