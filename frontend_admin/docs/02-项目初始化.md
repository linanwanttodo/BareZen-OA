## 2.2 配置文件

### 2.2.1 vite.config.ts

Vite 构建配置，包含路径别名、自动导入、代理等：

```typescript
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import AutoImport from 'unplugin-auto-import/vite'
import Components from 'unplugin-vue-components/vite'
import { ElementPlusResolver } from 'unplugin-vue-components/resolvers'
import { resolve } from 'path'

export default defineConfig({
  plugins: [
    vue(),
    AutoImport({
      // 自动导入 Vue 相关 API（ref, computed, watch 等）
      imports: ['vue', 'vue-router', 'pinia'],
      // 自动导入并注册组件（Element Plus 等）
      resolvers: [ElementPlusResolver()],
      // 生成自动导入的 TypeScript 声明文件
      dts: 'src/auto-imports.d.ts'
    }),
    Components({
      // 自动解析 Element Plus 组件
      resolvers: [ElementPlusResolver()],
      // 生成组件声明文件
      dts: 'src/components.d.ts'
    })
  ],
  resolve: {
    alias: {
      // 路径别名：@ 指向 src/
      '@': resolve(__dirname, 'src')
    }
  },
  server: {
    port: 5173,
    host: '0.0.0.0',
    proxy: {
      // 开发环境代理：/api -> 后端服务
      '/api': {
        target: 'http://localhost:8080',
        changeOrigin: true,
        // 不重写路径，保持 /api 前缀
        rewrite: (path) => path
      }
    }
  },
  build: {
    target: 'es2020',
    // 启用 gzip 压缩
    reportCompressedSize: true,
    // 分块策略
    rollupOptions: {
      output: {
        manualChunks: {
          'element-plus': ['element-plus'],
          'vue-vendor': ['vue', 'vue-router', 'pinia']
        }
      }
    }
  }
})
```

### 2.3.2 src/types/api.d.ts

```typescript
/**
 * 统一分页请求参数
 */
export interface PageQuery {
  pageNum: number
  pageSize: number
  keyword?: string
}

/**
 * 分页响应结构
 */
export interface PageResult<T> {
  rows: T[]
  total: number
  pageNum: number
  pageSize: number
}

/**
 * 统一响应结构（后端 Result<T> 对应）
 */
export interface ApiResult<T = unknown> {
  code: number
  msg: string
  data: T
}

/**
 * 操作结果（无数据返回）
 */
export type SimpleResult = ApiResult<null>

/**
 * 批量操作请求
 */
export interface BatchRequest {
  ids: number[]
}
```

## 2.5 入口文件

### 2.5.1 src/main.ts

应用入口：

```typescript
import { createApp } from 'vue'
import ElementPlus from 'element-plus'
import 'element-plus/dist/index.css'
import 'element-plus/theme-chalk/dark/css-vars.css'
import App from './App.vue'
import router from './router'
import { createPinia } from 'pinia'
import './styles/index.scss'

const app = createApp(App)

// 注册 Pinia 状态管理
app.use(createPinia())

// 注册 Vue Router
app.use(router)

// 注册 Element Plus
app.use(ElementPlus)

// 挂载到 #app
app.mount('#app')
```
