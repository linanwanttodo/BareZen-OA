## 14.2 操作日志页面 src/views/monitor/oper-log/index.vue

```vue
<template>
  <div class="page-container">
    <div class="search-bar">
      <el-form inline :model="queryParams" label-width="90px">
        <el-form-item label="操作模块">
          <el-input v-model="queryParams.title" placeholder="请输入操作模块" clearable />
        </el-form-item>
        <el-form-item label="操作类型">
          <el-select v-model="queryParams.businessType" placeholder="全部" clearable style="width: 150px">
            <el-option label="新增" :value="1" />
            <el-option label="修改" :value="2" />
            <el-option label="删除" :value="3" />
            <el-option label="查询" :value="4" />
            <el-option label="导出" :value="5" />
            <el-option label="导入" :value="6" />
            <el-option label="授权" :value="7" />
            <el-option label="其他" :value="0" />
          </el-select>
        </el-form-item>
        <el-form-item label="操作状态">
          <el-select v-model="queryParams.status" placeholder="全部" clearable style="width: 150px">
            <el-option label="成功" :value="0" />
            <el-option label="异常" :value="1" />
          </el-select>
        </el-form-item>
        <el-form-item label="操作时间">
          <el-date-picker
            v-model="dateRange"
            type="daterange"
            range-separator="至"
            start-placeholder="开始日期"
            end-placeholder="结束日期"
            value-format="YYYY-MM-DD"
            style="width: 240px"
          />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="handleSearch">搜索</el-button>
          <el-button @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </div>

    <div class="toolbar">
      <el-button type="danger" @click="handleClean" v-permission="['system:operlog:remove']">
        <el-icon><Delete /></el-icon> 清空日志
      </el-button>
    </div>

    <el-table v-loading="loading" :data="list" stripe>
      <el-table-column label="操作模块" prop="title" width="150" />
      <el-table-column label="操作类型" prop="businessType" width="100" align="center">
        <template #default="{ row }">
          <el-tag :type="businessTypeMap[row.businessType]">
            {{ businessTypeText[row.businessType] }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column label="操作方法" prop="method" min-width="200" show-overflow-tooltip />
      <el-table-column label="请求方式" prop="requestMethod" width="100" align="center">
        <template #default="{ row }">
          <el-tag size="small">{{ row.requestMethod }}</el-tag>
        </template>
      </el-table-column>
      <el-table-column label="操作人" prop="operator" width="120" />
      <el-table-column label="操作地址" prop="ip" width="130" />
      <el-table-column label="状态" prop="status" width="80" align="center">
        <template #default="{ row }">
          <el-tag :type="row.status === 0 ? 'success' : 'danger'" size="small">
            {{ row.status === 0 ? '成功' : '异常' }}
          </el-tag>
        </template>
      </el-table-column>
      <el-table-column label="操作时间" prop="operTime" width="170" />
      <el-table-column label="操作" width="100" fixed="right" align="center">
        <template #default="{ row }">
          <el-button type="primary" link @click="handleDetail(row)">详情</el-button>
        </template>
      </el-table-column>
    </el-table>

    <Pagination v-model:page="queryParams.pageNum" v-model:limit="queryParams.pageSize" :total="total" @pagination="loadData" />

    <!-- 详情弹窗 -->
    <el-dialog v-model="detailVisible" title="操作日志详情" width="700px">
      <el-descriptions :column="2" border>
        <el-descriptions-item label="操作模块">{{ currentLog?.title }}</el-descriptions-item>
        <el-descriptions-item label="操作类型">{{ businessTypeText[currentLog?.businessType || 0] }}</el-descriptions-item>
        <el-descriptions-item label="操作人">{{ currentLog?.operator }}</el-descriptions-item>
        <el-descriptions-item label="操作地址">{{ currentLog?.ip }}</el-descriptions-item>
        <el-descriptions-item label="操作时间">{{ currentLog?.operTime }}</el-descriptions-item>
        <el-descriptions-item label="状态">
          <el-tag :type="currentLog?.status === 0 ? 'success' : 'danger'">
            {{ currentLog?.status === 0 ? '成功' : '异常' }}
          </el-tag>
        </el-descriptions-item>
        <el-descriptions-item label="请求方法" :span="2">{{ currentLog?.requestMethod }} {{ currentLog?.method }}</el-descriptions-item>
        <el-descriptions-item label="请求参数" :span="2">
          <pre class="whitespace-pre-wrap text-xs text-gray-600">{{ currentLog?.operParam }}</pre>
        </el-descriptions-item>
        <el-descriptions-item v-if="currentLog?.errorMsg" label="错误信息" :span="2">
          <pre class="whitespace-pre-wrap text-xs text-red-500">{{ currentLog?.errorMsg }}</pre>
        </el-descriptions-item>
      </el-descriptions>
    </el-dialog>
  </div>
</template>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { Delete } from '@element-plus/icons-vue'
import Pagination from '@/components/Pagination.vue'
import { usePage } from '@/composables/usePage'
import { getOperLogList, cleanOperLogs } from '@/api/operLog'
import type { OperLogInfo, OperLogQuery } from '@/api/operLog'

const { loading, list, total, loadData } = usePage<OperLogInfo>()
const queryParams = reactive<OperLogQuery>({ pageNum: 1, pageSize: 10 })
const dateRange = ref<[string, string]>()
const detailVisible = ref(false)
const currentLog = ref<OperLogInfo | null>(null)

const businessTypeText: Record<number, string> = { 0: '其他', 1: '新增', 2: '修改', 3: '删除', 4: '查询', 5: '导出', 6: '导入', 7: '授权' }
const businessTypeMap: Record<number, string> = { 0: 'info', 1: 'success', 2: 'warning', 3: 'danger', 4: '', 5: 'warning', 6: 'success', 7: '' }

function handleSearch() { queryParams.pageNum = 1; loadData(getOperLogList) }
function handleReset() { Object.assign(queryParams, { pageNum: 1, pageSize: 10, title: '', businessType: undefined, status: undefined }); dateRange.value = undefined; loadData(getOperLogList) }
function handleDetail(row: OperLogInfo) { currentLog.value = row; detailVisible.value = true }

async function handleClean() {
  await ElMessageBox.confirm('确定清空所有操作日志吗？此操作不可恢复！', '警告', { type: 'warning' })
  await cleanOperLogs()
  ElMessage.success('清空成功')
  loadData(getOperLogList)
}

onMounted(() => { loadData(getOperLogList) })
</script>
```
