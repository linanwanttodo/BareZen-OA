# 05 - 请求封装

> **前置要求**：完成第 04 章状态管理。

## 5.1 为什么需要 Axios 封装？

直接使用 Axios 会导致大量重复代码。通过封装：
- **统一拦截**：请求自动带 Token，响应自动处理 401
- **统一错误处理**：后端返回的 `Result` 结构自动解析
- **类型安全**：TypeScript 类型提示
- **统一 URL 前缀**：环境变量控制

## 5.2 请求实例 src/api/request.ts

Axios 实例封装，包含请求/响应拦截器：

```typescript
import axios, { type AxiosInstance, type AxiosRequestConfig, type AxiosResponse, type InternalAxiosRequestConfig } from 'axios'
import { ElMessage, ElMessageBox } from 'element-plus'
import router from '@/router'
import { getToken } from '@/utils/auth'

// 创建 Axios 实例
const service: AxiosInstance = axios.create({
  // 后端基础路径
  baseURL: import.meta.env.VITE_APP_BASE_API,
  // 请求超时时间
  timeout: 10000,
  // 是否携带跨域 Cookie
  withCredentials: false,
  headers: {
    'Content-Type': 'application/json;charset=UTF-8'
  }
})

// ============ 请求拦截器 ============
service.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    // 如果是开发环境，打印请求信息
    if (import.meta.env.DEV) {
      console.log(`[API] ${config.method?.toUpperCase()} ${config.url}`, config.data || config.params)
    }

    // 自动注入 Token
    const token = getToken()
    if (token && config.headers) {
      config.headers['Authorization'] = `Bearer ${token}`
    }

    return config
  },
  (error) => {
    console.error('[API] Request Error:', error)
    return Promise.reject(error)
  }
)

// ============ 响应拦截器 ============
service.interceptors.response.use(
  (response: AxiosResponse) => {
    const res = response.data

    // 如果是开发环境，打印响应信息
    if (import.meta.env.DEV) {
      console.log(`[API] ${response.config.url} =>`, res)
    }

    // ============ 根据后端 Result 规范判断 ============
    // code === 200：业务成功
    // code === 401：未授权（Token 过期/无效）
    // code === 403：无权限
    // code === 500：服务端异常
    if (res.code === 200) {
      return res
    }

    // 业务错误：显示错误消息后返回 Rejected Promise
    ElMessage.error(res.msg || '操作失败')
    return Promise.reject(new Error(res.msg || '操作失败'))
  },
  (error) => {
    // 处理 HTTP 错误
    if (error.response) {
      const { status, data } = error.response

      switch (status) {
        case 401:
          // Token 过期：弹出确认框，用户确认后跳转登录
          ElMessageBox.confirm('登录已过期，请重新登录', '提示', {
            type: 'warning',
            confirmButtonText: '重新登录',
            cancelButtonText: '取消'
          }).then(() => {
            // 清除本地用户状态
            import('@/stores/user').then(({ useUserStore }) => {
              useUserStore().resetUserInfo()
            })
            router.push('/login')
          })
          break

        case 403:
          ElMessage.error('您没有权限访问此资源')
          break

        case 404:
          ElMessage.error('请求的资源不存在')
          break

        case 500:
          ElMessage.error(data?.msg || '服务器异常，请联系管理员')
          break

        default:
          ElMessage.error(data?.msg || `请求失败 (HTTP ${status})`)
      }
    } else if (error.request) {
      // 请求已发出但没有收到响应（通常是网络问题）
      ElMessage.error('网络连接失败，请检查网络')
    } else {
      // 请求配置出错
      ElMessage.error(error.message || '请求配置错误')
    }

    return Promise.reject(error)
  }
)

export default service
```

## 5.5 Composable 封装

### 5.5.1 分页逻辑 src/composables/usePage.ts

```typescript
import { ref, computed } from 'vue'
import type { PageQuery, PageResult } from '@/types/api'

/**
 * 分页逻辑封装
 * 所有列表页面都可以使用此 Hook
 */
export function usePage<T>() {
  const loading = ref(false)
  const pageNum = ref(1)
  const pageSize = ref(10)
  const total = ref(0)
  const list = ref<T[]>([]) as { value: T[] }

  const queryParams = computed<PageQuery>(() => ({
    pageNum: pageNum.value,
    pageSize: pageSize.value
  }))

  /**
   * 加载数据（调用者传入 fetcher 函数）
   */
  async function loadData(fetcher: (params: PageQuery) => Promise<PageResult<T>>) {
    loading.value = true
    try {
      const result = await fetcher(queryParams.value)
      list.value = result.rows
      total.value = result.total
    } finally {
      loading.value = false
    }
  }

  /**
   * 重置到第一页并刷新
   */
  async function refresh(fetcher: (params: PageQuery) => Promise<PageResult<T>>) {
    pageNum.value = 1
    await loadData(fetcher)
  }

  /**
   * 页码变化
   */
  function onPageChange(page: number) {
    pageNum.value = page
  }

  /**
   * 每页条数变化
   */
  function onSizeChange(size: number) {
    pageSize.value = size
    pageNum.value = 1
  }

  return {
    loading,
    list,
    pageNum,
    pageSize,
    total,
    queryParams,
    loadData,
    refresh,
    onPageChange,
    onSizeChange
  }
}
```

### 5.5.2 文件下载逻辑 src/composables/useDownload.ts

```typescript
import { ElMessage } from 'element-plus'

/**
 * 文件下载逻辑
 */
export function useDownload() {
  /**
   * 根据文件 ID 下载
   */
  function downloadById(id: number, fileName?: string) {
    const baseUrl = import.meta.env.VITE_APP_BASE_API
    const url = `${baseUrl}/system/v1/files/download/${id}`

    const token = localStorage.getItem('barezen_token')
    const a = document.createElement('a')
    a.href = `${url}?token=${token}`
    a.download = fileName || ''
    a.click()
  }

  /**
   * 根据 URL 下载（带 Token）
   */
  function downloadByUrl(url: string, fileName?: string) {
    const token = localStorage.getItem('barezen_token')
    const a = document.createElement('a')
    a.href = `${url}?token=${token}`
    a.download = fileName || ''
    a.click()
  }

  return {
    downloadById,
    downloadByUrl
  }
}
```
