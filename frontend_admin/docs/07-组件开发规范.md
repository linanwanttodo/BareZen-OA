## 7.1 组件分类

管理后台的组件分为两类：

| 类型 | 说明 | 存放位置 |
|:---|:---|:---|
| **全局通用组件** | 任何页面都可能用到（分页、图标选择器） | `src/components/` |
| **业务页面组件** | 仅本模块使用（用户表单、角色分配弹窗） | `src/views/{module}/components/` |

## 7.2 全局通用组件

### 7.2.1 分页组件 src/components/Pagination.vue

封装 Element Plus 分页，减少重复配置：

```vue
<template>
  <div class="pagination-container">
    <el-pagination
      v-model:current-page="currentPage"
      v-model:page-size="pageSize"
      :page-sizes="[10, 20, 50, 100]"
      :total="total"
      :background="true"
      :small="false"
      layout="total, sizes, prev, pager, next, jumper"
      @size-change="handleSizeChange"
      @current-change="handleCurrentChange"
    />
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'

interface Props {
  total: number
  page: number
  limit: number
}

interface Emits {
  (e: 'update:page', page: number): void
  (e: 'update:limit', limit: number): void
  (e: 'pagination', page: number, limit: number): void
}

const props = withDefaults(defineProps<Props>(), {
  total: 0,
  page: 1,
  limit: 10
})

const emit = defineEmits<Emits>()

const currentPage = computed({
  get: () => props.page,
  set: (val) => emit('update:page', val)
})

const pageSize = computed({
  get: () => props.limit,
  set: (val) => emit('update:limit', val)
})

function handleSizeChange(size: number) {
  emit('update:limit', size)
  emit('pagination', currentPage.value, size)
}

function handleCurrentChange(page: number) {
  emit('update:page', page)
  emit('pagination', page, pageSize.value)
}
</script>

<style lang="scss" scoped>
.pagination-container {
  display: flex;
  justify-content: flex-end;
  margin-top: 16px;
}
</style>
```

## 7.3 业务页面组件

### 7.3.1 用户表单弹窗 src/views/system/user/components/UserForm.vue

```vue
<template>
  <el-dialog
    v-model="visible"
    :title="isEdit ? '编辑用户' : '新增用户'"
    width="600px"
    @closed="handleClosed"
  >
    <el-form
      ref="formRef"
      :model="form"
      :rules="rules"
      label-width="100px"
    >
      <el-form-item label="用户名称" prop="userName">
        <el-input v-model="form.userName" placeholder="请输入用户名称" />
      </el-form-item>

      <el-form-item label="用户昵称" prop="nickName">
        <el-input v-model="form.nickName" placeholder="请输入用户昵称" />
      </el-form-item>

      <el-form-item label="手机号码" prop="phone">
        <el-input v-model="form.phone" placeholder="请输入手机号码" />
      </el-form-item>

      <el-form-item label="邮箱" prop="email">
        <el-input v-model="form.email" placeholder="请输入邮箱" />
      </el-form-item>

      <el-form-item label="性别">
        <el-radio-group v-model="form.sex">
          <el-radio :value="0">未知</el-radio>
          <el-radio :value="1">男</el-radio>
          <el-radio :value="2">女</el-radio>
        </el-radio-group>
      </el-form-item>

      <el-form-item v-if="!isEdit" label="用户密码" prop="password">
        <el-input
          v-model="form.password"
          type="password"
          placeholder="请输入密码"
          show-password
        />
      </el-form-item>

      <el-form-item label="状态">
        <el-radio-group v-model="form.status">
          <el-radio :value="0">正常</el-radio>
          <el-radio :value="1">停用</el-radio>
        </el-radio-group>
      </el-form-item>

      <el-form-item label="备注">
        <el-input v-model="form.remark" type="textarea" :rows="2" placeholder="请输入备注" />
      </el-form-item>
    </el-form>

    <template #footer>
      <el-button @click="visible = false">取消</el-button>
      <el-button type="primary" :loading="saving" @click="handleSubmit">确定</el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue'
import { ElMessage, type FormInstance, type FormRules } from 'element-plus'
import type { UserInfo } from '@/types/user'
import { addUser, updateUser } from '@/api/user'

interface Props {
  modelValue: boolean
  data?: UserInfo | null
}

interface Emits {
  (e: 'update:modelValue', val: boolean): void
  (e: 'success'): void
}

const props = defineProps<Props>()
const emit = defineEmits<Emits>()

const formRef = ref<FormInstance>()
const saving = ref(false)

const visible = computed({
  get: () => props.modelValue,
  set: (val) => emit('update:modelValue', val)
})

const isEdit = computed(() => !!props.data?.userId)

const form = ref({
  userName: '',
  nickName: '',
  phone: '',
  email: '',
  sex: 0,
  password: '',
  status: 0,
  remark: ''
})

const rules: FormRules = {
  userName: [
    { required: true, message: '请输入用户名称', trigger: 'blur' },
    { min: 2, max: 20, message: '长度为 2-20 个字符', trigger: 'blur' }
  ],
  nickName: [
    { required: true, message: '请输入用户昵称', trigger: 'blur' }
  ],
  phone: [
    { pattern: /^1[3-9]\d{9}$/, message: '请输入正确的手机号', trigger: 'blur' }
  ],
  email: [
    { type: 'email', message: '请输入正确的邮箱', trigger: 'blur' }
  ],
  password: [
    { required: true, message: '请输入密码', trigger: 'blur' },
    { min: 6, max: 20, message: '密码长度为 6-20 个字符', trigger: 'blur' }
  ]
}

// 监听 data 变化，自动填充表单
watch(
  () => props.data,
  (val) => {
    if (val) {
      Object.assign(form.value, val)
    } else {
      resetForm()
    }
  },
  { immediate: true }
)

function resetForm() {
  form.value = {
    userName: '',
    nickName: '',
    phone: '',
    email: '',
    sex: 0,
    password: '',
    status: 0,
    remark: ''
  }
  formRef.value?.resetFields()
}

function handleClosed() {
  resetForm()
}

async function handleSubmit() {
  const valid = await formRef.value?.validate().catch(() => false)
  if (!valid) return

  saving.value = true
  try {
    if (isEdit.value) {
      await updateUser(props.data!.userId, form.value)
      ElMessage.success('修改成功')
    } else {
      await addUser(form.value)
      ElMessage.success('新增成功')
    }
    emit('success')
    visible.value = false
  } finally {
    saving.value = false
  }
}
</script>
```

### 7.3.2 角色权限分配弹窗 src/views/system/role/components/RoleMenu.vue

```vue
<template>
  <el-dialog v-model="visible" title="分配权限" width="400px">
    <el-tree
      ref="treeRef"
      :data="menuList"
      :props="{ label: 'menuName', children: 'children' }"
      node-key="id"
      :default-checked-keys="checkedKeys"
      :default-expand-all="true"
      show-checkbox
      check-strictly
    />

    <template #footer>
      <el-button @click="visible = false">取消</el-button>
      <el-button type="primary" :loading="saving" @click="handleSubmit">确定</el-button>
    </template>
  </el-dialog>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue'
import { ElMessage } from 'element-plus'
import { getMenuTreeselect } from '@/api/menu'
import { getRoleMenus, assignMenus } from '@/api/role'
import type { ElTree } from 'element-plus'

interface Props {
  modelValue: boolean
  roleId?: number
}

interface Emits {
  (e: 'update:modelValue', val: boolean): void
  (e: 'success'): void
}

const props = defineProps<Props>()
const emit = defineEmits<Emits>()

const treeRef = ref<InstanceType<typeof ElTree>>()
const saving = ref(false)
const menuList = ref<any[]>([])
const checkedKeys = ref<number[]>([])

const visible = computed({
  get: () => props.modelValue,
  set: (val) => emit('update:modelValue', val)
})

watch(visible, async (val) => {
  if (val && props.roleId) {
    await loadMenus()
    await loadRoleMenus()
  }
})

async function loadMenus() {
  const { data } = await getMenuTreeselect()
  menuList.value = data || []
}

async function loadRoleMenus() {
  if (!props.roleId) return
  const { data } = await getRoleMenus(props.roleId)
  checkedKeys.value = getLeafKeys(data || [])
}

function getLeafKeys(menus: any[]): number[] {
  const keys: number[] = []
  menus.forEach((menu) => {
    if (!menu.children || menu.children.length === 0) {
      keys.push(menu.id)
    } else {
      keys.push(...getLeafKeys(menu.children))
    }
  })
  return keys
}

async function handleSubmit() {
  if (!props.roleId) return
  saving.value = true
  try {
    const menuIds = treeRef.value?.getCheckedKeys(false) as number[]
    await assignMenus(props.roleId, menuIds)
    ElMessage.success('分配成功')
    emit('success')
    visible.value = false
  } finally {
    saving.value = false
  }
}
</script>
```

## 7.4 页面开发规范

### 7.4.1 列表页面模板

每个管理后台的列表页面遵循统一结构：

```vue
<template>
  <div class="page-container">
    <!-- 搜索栏 -->
    <div class="search-bar">
      <el-form inline :model="queryParams">
        <el-form-item label="关键词">
          <el-input v-model="queryParams.keyword" placeholder="请输入关键词" clearable @keyup.enter="handleSearch" />
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="handleSearch">搜索</el-button>
          <el-button @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </div>

    <!-- 操作栏 -->
    <div class="toolbar">
      <el-button type="primary" v-permission="['system:user:add']" @click="handleAdd">
        新增
      </el-button>
    </div>

    <!-- 表格 -->
    <el-table v-loading="loading" :data="list" stripe>
      <el-table-column label="操作" width="180" fixed="right">
        <template #default="{ row }">
          <el-button type="primary" link v-permission="['system:user:edit']" @click="handleEdit(row)">编辑</el-button>
          <el-button type="danger" link v-permission="['system:user:del']" @click="handleDelete(row)">删除</el-button>
        </template>
      </el-table-column>
    </el-table>

    <!-- 分页 -->
    <Pagination
      v-model:page="queryParams.pageNum"
      v-model:limit="queryParams.pageSize"
      :total="total"
      @pagination="loadData"
    />
  </div>
</template>

<script setup lang="ts">
import { ref, reactive, onMounted } from 'vue'
import { ElMessageBox, ElMessage } from 'element-plus'
import Pagination from '@/components/Pagination.vue'
import { usePage } from '@/composables/usePage'
import { getUserList, deleteUser } from '@/api/user'
import type { UserInfo, UserQuery } from '@/types/user'

// 分页 + 列表数据
const { loading, list, total, loadData } = usePage<UserInfo>()

// 查询参数
const queryParams = reactive<UserQuery>({
  pageNum: 1,
  pageSize: 10,
  keyword: ''
})

function handleSearch() {
  queryParams.pageNum = 1
  loadData(getUserList)
}

function handleReset() {
  Object.assign(queryParams, { pageNum: 1, pageSize: 10, keyword: '' })
  loadData(getUserList)
}

async function handleDelete(row: UserInfo) {
  await ElMessageBox.confirm(`确定删除用户 "${row.nickName}" 吗？`, '提示', { type: 'warning' })
  await deleteUser(row.userId)
  ElMessage.success('删除成功')
  loadData(getUserList)
}

onMounted(() => {
  loadData(getUserList)
})
</script>
```
