## 18.2 接口设计

| 接口 | 方法 | 路径 | 说明 |
|:---|:---|:---|:---|
| 文章列表 | GET | `/api/app/v1/articles` | 分页 + 搜索 + 分类筛选 |
| 文章详情 | GET | `/api/app/v1/articles/{id}` | 详情 + 浏览量 +1 |
| 点赞 | POST | `/api/app/v1/articles/{id}/like` | 点赞数 +1 |
| 分类列表 | GET | `/api/app/v1/categories` | 全部分类 |
| 评论列表 | GET | `/api/app/v1/articles/{id}/comments` | 分页评论 |
| 发布评论 | POST | `/api/app/v1/articles/{id}/comments` | 需登录 |

## 18.4 前端对接

用户前台 `frontend_user` 的文章页面已对接真实 API：

```typescript
// src/api/content.ts
import request from './request'

export function getArticleList(params: ArticleQuery): Promise<ApiResult<PageResult<Article>>> {
  return request.get('/app/v1/articles', { params })
}

export function getArticle(id: number): Promise<ApiResult<Article>> {
  return request.get(`/app/v1/articles/${id}`)
}

export function likeArticle(id: number): Promise<ApiResult<null>> {
  return request.post(`/app/v1/articles/${id}/like`)
}

export function getCategories(): Promise<ApiResult<Category[]>> {
  return request.get('/app/v1/categories')
}

export function getComments(articleId: number, page = 1): Promise<ApiResult<PageResult<Comment>>> {
  return request.get(`/app/v1/articles/${articleId}/comments`, { params: { page } })
}

export function addComment(articleId: number, content: string): Promise<ApiResult<null>> {
  return request.post(`/app/v1/articles/${articleId}/comments`, { content })
}
```

## 18.6 启动后端 + 前端联调

```bash
# 1. 启动 MySQL + Redis
make env-up

# 2. 执行建表 SQL（按顺序）
mysql -u root -p barezen < backend/sql/init_rbac.sql
mysql -u root -p barezen < backend/sql/sys_article.sql

# 3. 启动后端
cd backend && ./gradlew :bz_web_app:bootRun

# 4. 启动管理后台
cd frontend_admin && pnpm dev

# 5. 启动用户前台
cd frontend_user && pnpm dev
```
