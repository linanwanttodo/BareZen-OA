# 04 - 状态管理

> **前置要求**：完成第 02 章项目初始化。

## 4.1 Pinia 核心概念

Pinia 是 Vue 官方推荐的新一代状态管理库，相比 Vuex 更轻量、更 TypeScript 友好。

**核心概念**：
- **Store**：存储状态和业务逻辑的容器（类似 Vuex 的 Module）
- **State**：响应式状态数据（类似 `data()`）
- **Getters**：计算属性（类似 `computed()`）
- **Actions**：修改状态的方法（类似 `mutations` + `actions`，但都是同步/异步均可）

## 4.2 用户状态 src/stores/user.ts

用户状态管理，负责用户信息和 Token：

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import type { UserInfo, LoginForm, LoginResult } from '@/types/user'
import { getToken, setToken, removeToken } from '@/utils/auth'

export const useUserStore = defineStore('user', () => {
  // ============ State ============
  const token = ref<string>(getToken() || '')
  const userId = ref<number>(0)
  const userName = ref<string>('')
  const nickName = ref<string>('')
  const avatar = ref<string>('')
  const roles = ref<string[]>([])
  const permissions = ref<string[]>([])

  // ============ Getters ============
  const isLoggedIn = computed(() => !!token.value)
  const userInfo = computed<UserInfo>(() => ({
    userId: userId.value,
    userName: userName.value,
    nickName: nickName.value,
    avatar: avatar.value
  }))

  // ============ Actions ============

  /**
   * 用户登录
   */
  async function login(loginForm: LoginForm): Promise<LoginResult> {
    const { login } = await import('@/api/auth')
    const result = await login(loginForm)

    // 保存 Token
    setToken(result.token)
    token.value = result.token

    // 保存用户信息
    setUserInfo(result)

    return result
  }

  /**
   * 获取当前用户信息
   */
  async function getUserInfo(): Promise<UserInfo> {
    const { getCurrentUser } = await import('@/api/auth')
    const result = await getCurrentUser()

    setUserInfo(result)
    return result
  }

  /**
   * 设置用户信息（从登录响应或用户信息接口获取）
   */
  function setUserInfo(info: UserInfo | LoginResult) {
    userId.value = info.userId
    userName.value = info.userName
    nickName.value = 'nickName' in info ? info.nickName : info.nickName

    if ('avatar' in info && info.avatar) {
      avatar.value = info.avatar
    }
  }

  /**
   * 退出登录
   */
  async function logout() {
    try {
      const { logout } = await import('@/api/auth')
      await logout()
    } finally {
      resetUserInfo()
    }
  }

  /**
   * 重置用户信息（清除本地状态）
   */
  function resetUserInfo() {
    token.value = ''
    userId.value = 0
    userName.value = ''
    nickName.value = ''
    avatar.value = ''
    roles.value = []
    permissions.value = []
    removeToken()
  }

  return {
    // State
    token,
    userId,
    userName,
    nickName,
    avatar,
    roles,
    permissions,
    // Getters
    isLoggedIn,
    userInfo,
    // Actions
    login,
    getUserInfo,
    logout,
    resetUserInfo
  }
})
```

## 4.3 权限状态 src/stores/permission.ts

管理动态路由和权限信息：

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'
import type { AppRouteRecordRaw } from '@/types/router'
import { buildDynamicRoutes } from '@/router/routes/async'
import { resetRouter } from '@/router'

export const usePermissionStore = defineStore('permission', () => {
  // ============ State ============
  /** 动态路由列表 */
  const routes = ref<AppRouteRecordRaw[]>([])
  /** 当前用户拥有的权限标识列表 */
  const permissions = ref<string[]>([])

  // ============ Actions ============

  /**
   * 获取并加载动态路由
   * 在路由守卫中登录成功后调用
   */
  async function loadRoutes(): Promise<AppRouteRecordRaw[]> {
    try {
      routes.value = await buildDynamicRoutes()
      return routes.value
    } catch (error) {
      console.error('加载动态路由失败:', error)
      return []
    }
  }

  /**
   * 设置权限标识
   */
  function setPermissions(perms: string[]) {
    permissions.value = perms
  }

  /**
   * 检查是否拥有指定权限
   */
  function hasPermission(perm: string | string[]): boolean {
    if (Array.isArray(perm)) {
      return perm.some((p) => permissions.value.includes(p))
    }
    return permissions.value.includes(perm)
  }

  /**
   * 重置权限状态
   */
  function resetPermission() {
    routes.value = []
    permissions.value = []
    resetRouter()
  }

  return {
    routes,
    permissions,
    loadRoutes,
    setPermissions,
    hasPermission,
    resetPermission
  }
})
```

## 4.4 路由缓存状态 src/stores/router.ts

管理已访问的路由标签页：

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'
import type { RouteLocationNormalized } from 'vue-router'

export const useRouterStore = defineStore('router', () => {
  // ============ State ============
  /** 已访问的路由列表（用于 TagsView） */
  const visitedRoutes = ref<RouteLocationNormalized[]>([])

  // ============ Actions ============

  /**
   * 添加已访问路由到标签栏
   */
  function addVisitedRoute(route: RouteLocationNormalized) {
    // 避免重复添加
    if (visitedRoutes.value.some((r) => r.path === route.path)) {
      return
    }
    visitedRoutes.value.push(route)
  }

  /**
   * 从标签栏移除指定路由
   */
  function removeVisitedRoute(path: string) {
    const index = visitedRoutes.value.findIndex((r) => r.path === path)
    if (index > -1) {
      visitedRoutes.value.splice(index, 1)
    }
  }

  /**
   * 清空所有已访问路由（保留首页）
   */
  function clearVisitedRoutes() {
    visitedRoutes.value = visitedRoutes.value.filter((r) => r.path === '/dashboard')
  }

  return {
    visitedRoutes,
    addVisitedRoute,
    removeVisitedRoute,
    clearVisitedRoutes
  }
})
```

## 4.5 设置状态 src/stores/settings.ts

全局设置（侧边栏折叠、多标签栏等）：

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'

export const useSettingsStore = defineStore('settings', () => {
  // ============ State ============
  const isCollapse = ref<boolean>(false)
  const isTagsView = ref<boolean>(true)
  const isFixedHeader = ref<boolean>(true)
  const showLogo = ref<boolean>(true)

  // ============ Actions ============

  function toggleSidebar() {
    isCollapse.value = !isCollapse.value
  }

  function setSidebar(val: boolean) {
    isCollapse.value = val
  }

  return {
    isCollapse,
    isTagsView,
    isFixedHeader,
    showLogo,
    toggleSidebar,
    setSidebar
  }
})
```

## 4.6 Token 工具函数 src/utils/auth.ts

Token 的存取操作（使用 Cookies）：

```typescript
import Cookies from 'js-cookie'

const TokenKey = 'barezen_token'
const expires = 7 // 7 天过期

export function getToken(): string | undefined {
  return Cookies.get(TokenKey)
}

export function setToken(token: string): void {
  Cookies.set(TokenKey, token, { expires })
}

export function removeToken(): void {
  Cookies.remove(TokenKey)
}
```

> **安装 js-cookie**：
> ```bash
> pnpm add js-cookie @types/js-cookie
> ```
