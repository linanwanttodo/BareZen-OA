## 7.2 首页主入口 src/pages/home/index.vue

```vue
<template>
  <div>
    <!-- 英雄区 Banner -->
    <HeroBanner />

    <!-- 分类导航 -->
    <div class="bg-white border-b border-gray-100">
      <div class="container mx-auto px-4">
        <CategoryNav :categories="categories" />
      </div>
    </div>

    <!-- 文章列表 -->
    <div class="container mx-auto px-4 py-10">
      <ArticleGrid :articles="articles" :loading="loading" @load-more="loadMore" />
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import HeroBanner from './components/HeroBanner.vue'
import CategoryNav from './components/CategoryNav.vue'
import ArticleGrid from './components/ArticleGrid.vue'
import { getArticleList, getCategories } from '@/api/content'

const loading = ref(false)
const articles = ref<any[]>([])
const categories = ref<any[]>([])
const page = ref(1)
const hasMore = ref(true)

async function loadArticles() {
  loading.value = true
  try {
    const { data } = await getArticleList({ page: page.value, pageSize: 12 })
    if (page.value === 1) {
      articles.value = data.rows
    } else {
      articles.value.push(...data.rows)
    }
    hasMore.value = data.rows.length === 12
  } finally {
    loading.value = false
  }
}

async function loadMore() {
  page.value++
  await loadArticles()
}

onMounted(async () => {
  await Promise.all([loadArticles(), loadCategories()])
})
</script>
```

## 7.4 CategoryNav 组件

```vue
<template>
  <div class="flex items-center gap-1 py-3 overflow-x-auto scrollbar-hide">
    <button
      class="px-4 py-1.5 rounded-full text-sm font-medium whitespace-nowrap"
      :class="activeCategory === 0 ? 'bg-primary-600 text-white' : 'text-gray-600 bg-gray-100'"
      @click="handleSelect(0)"
    >
      全部
    </button>
    <button
      v-for="cat in categories"
      :key="cat.id"
      class="px-4 py-1.5 rounded-full text-sm font-medium whitespace-nowrap"
      :class="activeCategory === cat.id ? 'bg-primary-600 text-white' : 'text-gray-600 bg-gray-100'"
      @click="handleSelect(cat.id)"
    >
      {{ cat.name }}
    </button>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'

interface Props {
  categories: any[]
}

const props = defineProps<Props>()
const emit = defineEmits(['select'])

const activeCategory = ref(0)

function handleSelect(id: number) {
  activeCategory.value = id
  emit('select', id)
}
</script>
```
