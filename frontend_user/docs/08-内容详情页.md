```vue
<template>
  <div class="container mx-auto px-4 py-10 max-w-4xl">
    <div v-if="article" class="space-y-8">
      <!-- 返回按钮 -->
      <el-button text @click="router.back()">
        <el-icon class="mr-1"><ArrowLeft /></el-icon> 返回
      </el-button>

      <!-- 文章头部 -->
      <header>
        <!-- 分类 + 标签 -->
        <div class="flex flex-wrap items-center gap-2 mb-4">
          <span class="px-3 py-1 bg-primary-100 text-primary-700 text-sm rounded-full">
            {{ article.category?.name }}
          </span>
          <span
            v-for="tag in article.tags"
            :key="tag"
            class="px-2 py-1 bg-gray-100 text-gray-600 text-xs rounded-full"
          >
            {{ tag }}
          </span>
        </div>

        <!-- 标题 -->
        <h1 class="text-3xl md:text-4xl font-bold text-gray-900 mb-4">
          {{ article.title }}
        </h1>

        <!-- 作者信息 -->
        <div class="flex items-center gap-4">
          <el-avatar :size="48" :src="article.author?.avatar">
            {{ article.author?.nickname?.charAt(0) }}
          </el-avatar>
          <div>
            <div class="font-medium text-gray-800">{{ article.author?.nickname }}</div>
            <div class="text-sm text-gray-400">
              {{ formatDate(article.createdAt) }} · {{ article.viewCount }} 阅读
            </div>
          </div>

          <!-- 点赞 -->
          <div class="ml-auto flex items-center gap-2">
            <button
              class="flex items-center gap-1 px-3 py-1.5 rounded-full border"
              :class="isLiked ? 'border-red-300 bg-red-50 text-red-500' : 'border-gray-200 text-gray-500'"
              @click="handleLike"
            >
              <span class="text-lg">{{ isLiked ? '' : '' }}</span>
              <span class="text-sm font-medium">{{ article.likeCount }}</span>
            </button>
          </div>
        </div>
      </header>

      <!-- 封面图 -->
      <div v-if="article.coverImage" class="rounded-xl overflow-hidden">
        <img :src="article.coverImage" :alt="article.title" class="w-full" />
      </div>

      <!-- 文章正文 (使用 v-html 渲染 Markdown/富文本) -->
      <article class="prose max-w-none" v-html="article.content" />

      <!-- 分割线 -->
      <hr class="border-gray-200" />

      <!-- 评论区 -->
      <section>
        <h3 class="text-lg font-semibold mb-4">
          评论 ({{ comments.total }})
        </h3>

        <!-- 发布评论 -->
        <div v-if="userStore.isLoggedIn" class="flex gap-3 mb-6">
          <el-avatar :size="36" :src="userStore.avatar">
            {{ userStore.nickname?.charAt(0) }}
          </el-avatar>
          <div class="flex-1">
            <el-input
              v-model="commentContent"
              type="textarea"
              :rows="3"
              placeholder="发表你的看法..."
            />
            <div class="flex justify-end mt-2">
              <el-button type="primary" size="small" @click="submitComment">
               发表评论
              </el-button>
            </div>
          </div>
        </div>
        <div v-else class="mb-6 p-4 bg-gray-50 rounded-lg text-center text-gray-500 text-sm">
          <router-link to="/auth/login" class="text-primary-600">登录</router-link> 后发表评论
        </div>

        <!-- 评论列表 -->
        <div class="space-y-4">
          <div v-for="comment in comments.list" :key="comment.id" class="flex gap-3">
            <el-avatar :size="36" :src="comment.user?.avatar">
              {{ comment.user?.nickname?.charAt(0) }}
            </el-avatar>
            <div class="flex-1">
              <div class="flex items-center gap-2">
                <span class="font-medium text-gray-800">{{ comment.user?.nickname }}</span>
                <span class="text-xs text-gray-400">{{ formatDate(comment.createdAt) }}</span>
              </div>
              <p class="mt-1 text-gray-600">{{ comment.content }}</p>
            </div>
          </div>
        </div>
      </section>
    </div>

    <!-- 加载状态 -->
    <div v-else-if="loading" class="flex justify-center py-20">
      <el-icon class="is-loading text-4xl text-primary-500"><Loading /></el-icon>
    </div>

    <!-- 错误/未找到 -->
    <div v-else class="text-center py-20 text-gray-400">
      <div class="text-5xl mb-4"></div>
      <p>文章不存在或已被删除</p>
      <el-button class="mt-4" @click="router.push('/')">返回首页</el-button>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { ElMessage } from 'element-plus'
import { ArrowLeft, Loading } from '@element-plus/icons-vue'
import { getArticle, likeArticle, getCommentList, addComment } from '@/api/content'
import { useUserStore } from '@/stores/user'

const route = useRoute()
const router = useRouter()
const userStore = useUserStore()

const loading = ref(false)
const article = ref<any>(null)
const isLiked = ref(false)
const commentContent = ref('')
const comments = ref({ list: [] as any[], total: 0 })

async function loadArticle() {
  loading.value = true
  try {
    const { data } = await getArticle(route.params.id as unknown as number)
    article.value = data
  } finally {
    loading.value = false
  }
}

async function loadComments() {
  const { data } = await getCommentList(route.params.id as unknown as number)
  comments.value = {
    list: data.rows || [],
    total: data.total || 0
  }
}

async function handleLike() {
  if (!userStore.isLoggedIn) {
    router.push('/auth/login')
    return
  }
  await likeArticle(article.value.id)
  isLiked.value = !isLiked.value
  article.value.likeCount += isLiked.value ? 1 : -1
}

async function submitComment() {
  if (!commentContent.value.trim()) return
  await addComment(article.value.id, commentContent.value)
  ElMessage.success('评论成功')
  commentContent.value = ''
  loadComments()
}

function formatDate(dateStr?: string): string {
  if (!dateStr) return ''
  return new Date(dateStr).toLocaleDateString('zh-CN', {
    year: 'numeric',
    month: 'short',
    day: 'numeric'
  })
}

onMounted(async () => {
  await loadArticle()
  await loadComments()
})
</script>

<style>

.prose h1 { @apply text-2xl font-bold mt-8 mb-4; }
.prose h2 { @apply text-xl font-bold mt-6 mb-3; }
.prose h3 { @apply text-lg font-bold mt-4 mb-2; }
.prose p { @apply text-gray-700 leading-relaxed mb-4; }
.prose img { @apply rounded-lg max-w-full; }
.prose pre { @apply bg-gray-800 text-gray-100 p-4 rounded-lg overflow-x-auto my-4; }
.prose code { @apply bg-gray-100 px-1.5 py-0.5 rounded text-sm; }
.prose pre code { @apply bg-transparent p-0; }
.prose blockquote { @apply border-l-4 border-primary-400 pl-4 italic text-gray-500 my-4; }
.prose ul { @apply list-disc pl-6 my-2; }
.prose ol { @apply list-decimal pl-6 my-2; }
.prose a { @apply text-primary-600 underline; }
</style>
```
