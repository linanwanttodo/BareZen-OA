# 04 - 状态管理

> **前置要求**：完成第 02 章项目初始化。

## 4.1 Pinia Store 设计

用户前台的状态管理相比管理后台更简单，主要关注：

| Store | 职责 |
|:---|:---|
| `useUserStore` | 用户登录状态、Token、个人信息 |
| `useAuthStore` | 认证相关（登录表单状态、验证码等） |
| `useCartStore` | 购物车（可选，如需电商功能） |

## 4.2 用户状态 src/stores/user.ts

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getToken, setToken, removeToken } from '@/utils/auth'

export const useUserStore = defineStore('user', () => {
  const token = ref<string>(getToken() || '')
  const userId = ref<number>(0)
  const username = ref<string>('')
  const nickname = ref<string>('')
  const email = ref<string>('')
  const phone = ref<string>('')
  const avatar = ref<string>('')
  const createdAt = ref<string>('')

  const isLoggedIn = computed(() => !!token.value)

  async function login(form: { username: string; password: string }) {
    const { login } = await import('@/api/auth')
    const result = await login(form)
    setToken(result.token)
    token.value = result.token
    setUserInfo(result.user)
    return result
  }

  function setUserInfo(user: any) {
    userId.value = user.id
    username.value = user.username
    nickname.value = user.nickname
    email.value = user.email || ''
    phone.value = user.phone || ''
    avatar.value = user.avatar || ''
    createdAt.value = user.createdAt || ''
  }

  async function logout() {
    try {
      const { logout } = await import('@/api/auth')
      await logout()
    } finally {
      resetUserInfo()
    }
  }

  function resetUserInfo() {
    token.value = ''
    userId.value = 0
    username.value = ''
    nickname.value = ''
    email.value = ''
    phone.value = ''
    avatar.value = ''
    createdAt.value = ''
    removeToken()
  }

  return {
    token,
    userId,
    username,
    nickname,
    email,
    phone,
    avatar,
    createdAt,
    isLoggedIn,
    login,
    setUserInfo,
    logout,
    resetUserInfo
  }
})
```

## 4.3 认证状态 src/stores/auth.ts

```typescript
import { defineStore } from 'pinia'
import { ref } from 'vue'

export const useAuthStore = defineStore('auth', () => {
  const captchaBase64 = ref('')
  const captchaUuid = ref('')

  async function fetchCaptcha() {
    const { getCaptcha } = await import('@/api/auth')
    const result = await getCaptcha()
    captchaBase64.value = result.captchaBase64
    captchaUuid.value = result.uuid
  }

  function resetCaptcha() {
    captchaBase64.value = ''
    captchaUuid.value = ''
  }

  return {
    captchaBase64,
    captchaUuid,
    fetchCaptcha,
    resetCaptcha
  }
})
```

## 4.4 Token 工具 src/utils/auth.ts

```typescript
import Cookies from 'js-cookie'

const TokenKey = 'barezen_token'
const expires = 7

export function getToken(): string | undefined {
  return Cookies.get(TokenKey)
}

export function setToken(token: string): void {
  Cookies.set(TokenKey, token, { expires })
}

export function removeToken(): void {
  Cookies.remove(TokenKey)
}
```
